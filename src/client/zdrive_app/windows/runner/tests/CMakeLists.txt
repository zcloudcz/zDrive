cmake_minimum_required(VERSION 3.14)
project(tray_window_test LANGUAGES CXX)
enable_testing()
foreach(target tray_window_test tray_window_no_icon_test)
  add_executable(${target} tray_window_test.cpp ../tray_window.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
  target_include_directories(${target} PRIVATE ..)
  target_link_libraries(${target} PRIVATE user32 shell32)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
target_sources(tray_window_test PRIVATE ../Runner.rc)
